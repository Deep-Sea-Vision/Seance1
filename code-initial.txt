/* ============================ ENGINE ============================ */

function mediaBlock(media){
  if(!media) return "";
  if(media.img) return `<img src="${media.img}" alt="">`;
  if(media.video) return `<video autoplay muted loop playsinline><source src="${media.video}"></video>`;
  return `<span>${media.caption || ""}</span>`;
}

function accentColor(sub){
  return { bluerov:"var(--bluerov)", military:"var(--military)", titan:"var(--titan)" }[sub] || "var(--ink)";
}

function renderHUD(scene){
  const hud = document.getElementById('hud');
  if(scene.kind === "menu"){ hud.classList.remove('on'); return; }
  hud.classList.add('on');
  document.getElementById('hudSub').textContent = scene.subLabel;
  document.getElementById('hudStat1').innerHTML = `DEPTH <b>${scene.depth}m</b>`;
  document.getElementById('hudStat2').innerHTML = `${scene.stat2Label} <b>${scene.stat2}</b>`;
  document.getElementById('hudBar').style.background = accentColor(scene.sub);
  hud.style.color = accentColor(scene.sub);
}

function tagFor(kind){
  if(kind === "warn") return `<div class="tag warn">DETECTION ERROR — MINOR</div>`;
  if(kind === "fail") return `<div class="tag fail">MISSION FAILED</div>`;
  if(kind === "success") return `<div class="tag good">MISSION COMPLETE</div>`;
  return "";
}

function render(id){
  const scene = SCENES[id];
  const stage = document.getElementById('stage');
  renderHUD(scene);

  if(scene.kind === "menu"){
    stage.innerHTML = `<div class="scene active">${scene.render()}</div>`;
    return;
  }

  const frameClass = scene.sub || "neutral";
  const mediaHTML = scene.media ? `<div class="media">${mediaBlock(scene.media)}</div>` : "";

  let choicesHTML = "";
  if(scene.kind === "warn"){
    choicesHTML = `<div class="choices">
      <button class="choice" onclick="goTo('${scene.backTo}')"><span class="marker">↩</span>Go back to the previous choice</button>
    </div>`;
  } else if(scene.kind === "fail"){
    choicesHTML = `<div class="choices">
      ${scene.choices ? scene.choices.map(c=>`<button class="choice" onclick="goTo('${c.next}')"><span class="marker">›</span>${c.label}</button>`).join("") : ""}
      ${scene.restartTo ? `<button class="choice" onclick="goTo('${scene.restartTo}')"><span class="marker">⟲</span>Restart the mission</button>` : ""}
    </div>`;
  } else {
    choicesHTML = `<div class="choices">
      ${scene.choices.map(c=>`<button class="choice" onclick="goTo('${c.next}')"><span class="marker">›</span>${c.label}</button>`).join("")}
    </div>`;
  }

  stage.innerHTML = `
    <div class="scene active">
      <div class="frame ${frameClass}">
        <span class="c2"></span><span class="c3"></span>
        ${tagFor(scene.kind)}
        <div class="eyebrow mono">${scene.subLabel}</div>
        <h1 class="scene-title">${scene.title}</h1>
        ${mediaHTML}
        <div class="narrative">${scene.body.map(p=>`<p>${p}</p>`).join("")}</div>
        ${choicesHTML}
      </div>
    </div>
  `;
  window.scrollTo(0,0);
}

function goTo(id){ render(id); }

render('menu');
